```python
"""What one request did inside the agent, for Agent Lab: LLM chains, knowledge base (IDP) and bank systems (SBE),
keyed by the request's x-trace-id. Kept in memory of this local process; the bank code is not changed."""
import json
from collections import OrderedDict
from collections.abc import Awaitable, Callable

import httpx

MAX_TRACES = 300
MAX_TEXT = 20_000

TraceId = Callable[[], str | None]
Hook = Callable[[httpx.Response], Awaitable[None]]


class Traces:
    def __init__(self, limit: int = MAX_TRACES) -> None:
        self._items: OrderedDict[str, dict] = OrderedDict()
        self._limit = limit

    def of(self, trace_id: str) -> dict:
        if trace_id not in self._items:
            self._items[trace_id] = {"traceId": trace_id, "chains": [], "rag": [], "systems": []}
            while len(self._items) > self._limit:
                self._items.popitem(last=False)
        return self._items[trace_id]

    def get(self, trace_id: str) -> dict | None:
        return self._items.get(trace_id)

    def __len__(self) -> int:
        return len(self._items)


def idp_hook(traces: Traces, trace_id: TraceId) -> Hook:
    async def record(response: httpx.Response) -> None:
        key = trace_id()
        if not key:
            return
        await response.aread()
        call = rag_call(_json(response.request.content), _json(response.content), response.status_code)
        traces.of(key)["rag"].append(call)

    return record


def sbe_hook(traces: Traces, trace_id: TraceId) -> Hook:
    async def record(response: httpx.Response) -> None:
        key = trace_id()
        if not key:
            return
        body = _json(response.request.content)
        tool = body.get("name") or body.get("tool") or response.request.url.path
        traces.of(key)["systems"].append(
            {"tool": tool, "arguments": body.get("arguments", body), "status": response.status_code})

    return record


def rag_call(request: dict, answer: dict, status: int) -> dict:
    asked = (request.get("message") or {}).get("request") or {}
    question = (asked.get("messages") or [{}])[-1]
    qa = ((request.get("configuration") or {}).get("agent_configuration") or {}).get("qa") or {}
    replies = (((answer.get("result") or {}).get("response") or {}).get("messages")) or []
    last = replies[-1] if replies else {}
    return {
        "query": question.get("content", ""),
        "filter": (asked.get("filter") or {}).get("value"),
        "systemPrompt": ((qa.get("prompts") or {}).get("stuff_system_prompt") or "") if qa.get("enabled") else "",
        "passages": [_passage(source) for source in last.get("sources") or []],
        "answer": (last.get("content") or "")[:MAX_TEXT],
        "reason": last.get("message_reason") or answer.get("message_reason"),
        "status": status,
    }


def _passage(source: dict) -> dict:
    meta = source.get("metadata") or {}
    # The scores sit beside the passage in IDP's answer, inside metadata in some of its versions.
    def score(key: str) -> float | None:
        return source.get(key, meta.get(key))

    return {
        "article": meta.get("sberhelp_uuid") or meta.get("index"),
        "passage": meta.get("passage_index"),
        "text": (source.get("content") or "")[:MAX_TEXT],
        "retrieval": score("retrieval_relevance_score"),
        "reranker": score("reranker_relevance_score"),
    }


def _json(content: bytes) -> dict:
    try:
        value = json.loads(content or b"{}")
    except ValueError:
        return {}
    return value if isinstance(value, dict) else {}
```

```python
class ChainRecorder(AsyncCallbackHandler):
    """Every model call of the request: the chain it belongs to, the messages as rendered, the model's answer."""

    run_inline = True  # in the caller's context, so the request's trace id is the caller's

    def __init__(self, traces: Traces, trace_id: TraceId) -> None:
        self._traces = traces
        self._trace_id = trace_id
        self._runs: dict[UUID, tuple[str, UUID | None]] = {}
        self._calls: dict[UUID, tuple[dict, float]] = {}

    async def on_chain_start(self, serialized: Any, inputs: Any, *, run_id: UUID,
                             parent_run_id: UUID | None = None, **kwargs: Any) -> None:
        self._runs[run_id] = (kwargs.get("name") or "", parent_run_id)

    async def on_chain_end(self, outputs: Any, *, run_id: UUID, **kwargs: Any) -> None:
        self._runs.pop(run_id, None)

    async def on_chain_error(self, error: BaseException, *, run_id: UUID, **kwargs: Any) -> None:
        self._runs.pop(run_id, None)

    async def on_chat_model_start(self, serialized: Any, messages: list[list[BaseMessage]], *, run_id: UUID,
                                  parent_run_id: UUID | None = None, **kwargs: Any) -> None:
        key = self._trace_id()
        if not key:
            return
        call = {"name": self._chain_name(parent_run_id), "messages": [_message(m) for m in messages[0]],
                "output": None}
        self._traces.of(key)["chains"].append(call)
        self._calls[run_id] = (call, time.monotonic())

    async def on_llm_end(self, response: LLMResult, *, run_id: UUID, **kwargs: Any) -> None:
        found = self._calls.pop(run_id, None)
        if found is not None:
            call, started = found
            call.update(output=_output(response), seconds=round(time.monotonic() - started, 2))

    async def on_llm_error(self, error: BaseException, *, run_id: UUID, **kwargs: Any) -> None:
        found = self._calls.pop(run_id, None)
        if found is not None:
            call, started = found
            call.update(error=repr(error), seconds=round(time.monotonic() - started, 2))

    def _chain_name(self, run_id: UUID | None) -> str:
        while run_id is not None and run_id in self._runs:
            name, parent = self._runs[run_id]
            if name and not name.startswith(PLUMBING):
                return name
            run_id = parent
        return UNNAMED
```

```python
def _message(message: BaseMessage) -> dict:
    content = message.content if isinstance(message.content, str) else json.dumps(message.content, ensure_ascii=False)
    return {"role": message.type, "content": content[:MAX_TEXT]}


def _output(response: LLMResult) -> str:
    generation = response.generations[0][0]
    text = generation.text
    message = getattr(generation, "message", None)
    if not text and message is not None:
        # Structured output answers with a tool call and no text.
        calls = getattr(message, "tool_calls", None) or message.additional_kwargs.get("function_call")
        text = json.dumps(calls, ensure_ascii=False, default=str) if calls else ""
    return text[:MAX_TEXT]
```

```python
def _add_response_hook(client, hook):
    hooks = client.event_hooks
    client.event_hooks = {**hooks, 'response': [*hooks.get('response', []), hook]}
```

```python
    reply, trace = step['reply'], step.get('trace')
    shown_history = [
        {'role': 'CUSTOMER', 'text': m['text']} if m['role'] == 'customer' else {'role': 'AGENT', 'text': logs.as_seen(m['text'])}
        for m in step['history']
    ]
    payload = {
        'expectations': rules,
        'history': shown_history,
        'customerMessage': step['customer'],
        'agentReply': {'text': reply['text'], 'status': reply['status'], 'buttons': reply.get('options') or []},
        'trace': rag.for_judge(trace),
    }
    answer = await llm.structured(JUDGE_REPLAY, payload, parse=lambda value: _parse_reply(value, rules), endpoint=endpoint)
    agent_text = '\n'.join([reply['text'], *(reply.get('options') or [])])
    rows = checked(
        answer.value.rules,
        rules,
        agent_text,
        tools=rag.tools(trace),
        knowledge_available=rag.called(trace),
        rag=rag.evidence(trace),
    )
    return Verdict(rows, verdict_of(rows), answer.model)
```

```python
"""Replay of exported conversations through the local agent: a step for every customer message, the history taken from
the logs, the agent's trace on every step, a verdict on the new reply (spec 2026-10-05-voice360-replay-design.md)."""

from . import discover, logs, rag, tone

RESULT = 'replay.json'
FAMILIES = ('tone', 'code', 'rag')


def steps(dialogue: dict) -> list[dict]:
    """A step for every customer message: the log before it, the message, and production's reply to it, if any."""
    said = [
        {'role': 'customer' if m['role'] == 'user' else 'agent', 'text': m['content']} for m in dialogue['messages']
    ]
    out = []
    for index, message in enumerate(said):
        if message['role'] != 'customer':
            continue
        following = said[index + 1] if index + 1 < len(said) else None
        reply = following['text'] if following and following['role'] == 'agent' else None
        out.append({
            'index': len(out),
            'history': said[:index],
            'customer': message['text'],
            'prodReply': logs.as_seen(reply) if reply is not None else None,
        })
    return out


def of_family(family: str, rules: list[dict]) -> list[dict]:
    """A check's rules under ids of their own family: tone and accuracy number their rules independently."""
    return [{**rule, 'id': f'{family}:{rule["id"]}', 'family': family} for rule in rules]


async def criteria_by_dialogue(dialogues: list[dict]) -> dict[str, list[dict]]:
    """Every dialogue's criteria: tone of voice, its accuracy topic's rules, the knowledge-base ones."""
    tone_rules = _tone_rules()
    accuracy = store.load(discover.RESULT) or {}
    topics = await discover.keep_topics(accuracy, dialogues) if accuracy.get('topics') else []
    topic_of = {dialogue_id: topic for topic in topics for dialogue_id in topic['dialogueIds']}
    return {
        str(d['id']): [
            *tone_rules,
            *of_family('code', (topic_of.get(str(d['id'])) or {}).get('rules') or []),
            *rag.CRITERIA,
        ]
        for d in dialogues
    }
```

```python
async def run(
    target: str,
    count: int,
    progress: Progress,
    *,
    create: Callable[[str], agents.HttpAgent] = agents.create,
    verdict: StepJudge = judge.step_verdict,
) -> dict:
    dialogues = discover.sample(count)
    if not dialogues:
        raise RuntimeError(NO_DIALOGUES)
    agent = create(target)
    if not agent.mocked:
        raise RuntimeError(NOT_LOCAL)
    progress(done=0, total=0, message='Собираем критерии')
    criteria = await criteria_by_dialogue(dialogues)
    started = store.now()
    plans = [(dialogue, steps(dialogue)) for dialogue in dialogues]
    counter = _Counter(sum(len(planned) for _, planned in plans), progress)
    gate = asyncio.Semaphore(PARALLEL)
    async with session(agent):

        async def one(dialogue: dict, planned: list[dict]) -> dict:
            async with gate:
                rules = criteria[str(dialogue['id'])]
                return await _replay_dialogue(agent, dialogue, planned, rules, verdict, counter)

        async with asyncio.TaskGroup() as tasks:
            running = [tasks.create_task(one(dialogue, planned)) for dialogue, planned in plans]
    played = [task.result() for task in running]
    value = {
        'id': uuid.uuid4().hex,
        'target': target,
        'version': agent.version,
        'startedAt': started,
        'finishedAt': store.now(),
        'model': llm.models_used([step for d in played for step in d['steps']]),
        'metric': metric(played),
        'dialogues': played,
    }
    store.save(RESULT, value)
    return value


def summary() -> dict | None:
    """What the state says of the latest replay: enough to know it changed."""
    value = store.load(RESULT)
    return {'id': value['id'], 'finishedAt': value['finishedAt']} if value else None


def metric(dialogues: list[dict]) -> dict:
    """PASS / (PASS + FAIL) over the steps' verdicts, per family of criteria."""
    counts = {family: {'pass': 0, 'fail': 0} for family in FAMILIES}
    for row in (row for d in dialogues for step in d['steps'] for row in step.get('rules') or []):
        family = counts.get(row['ruleId'].split(':', 1)[0])
        if family is not None and row['status'] in ('PASS', 'FAIL'):
            family['pass' if row['status'] == 'PASS' else 'fail'] += 1
    return {
        family: {**c, 'accuracy': c['pass'] / (c['pass'] + c['fail']) if c['pass'] + c['fail'] else None}
        for family, c in counts.items()
    }


def dialogue_status(played: list[dict]) -> str:
    statuses = {step.get('status') for step in played}
    if 'FAIL' in statuses:
        return 'FAIL'
    return 'PASS' if statuses == {'PASS'} else 'UNMEASURED'


class _Counter:
    def __init__(self, total: int, progress: Progress) -> None:
        self.done, self.total, self.progress = 0, total, progress
        progress(done=0, total=total, message='Повторяем разговоры')

    def step_done(self) -> None:
        self.done += 1
        self.progress(done=self.done, total=self.total, message='Повторяем разговоры')


async def _replay_dialogue(
    agent: agents.HttpAgent, dialogue: dict, planned: list[dict], rules: list[dict], verdict: StepJudge, counter: _Counter
) -> dict:
    conversation_id = str(uuid.uuid4())
    for step in planned:
        if await _play_step(agent, conversation_id, step):
            await _judge_step(rules, step, verdict)
        counter.step_done()
    return {'dialogueId': str(dialogue['id']), 'status': dialogue_status(planned), 'steps': planned}


async def _play_step(agent: agents.HttpAgent, conversation_id: str, step: dict) -> bool:
    try:
        reply = await agent.say(conversation_id, step['customer'], history=step['history'])
    except AgentError as error:
        step.update(status='UNMEASURED', error=str(error))
        return False
    if reply.get('trace') is None:
        raise RuntimeError(NO_TRACE)
    step['reply'] = {key: reply.get(key) for key in ('text', 'status', 'options', 'seconds')}
    step['trace'] = reply['trace']
    return True


async def _judge_step(rules: list[dict], step: dict, verdict: StepJudge) -> None:
    rag_called = rag.called(step['trace'])
    asked = [rule for rule in rules if rule['family'] != 'rag' or rag_called]
    skipped = [rag.skipped(rule) for rule in rules if rule['family'] == 'rag' and not rag_called]
    if not asked:
        step.update(rules=skipped, status=judge.verdict_of(skipped), error=None)
        return
    try:
        result = await verdict(asked, step)
        second = await judge.second_opinion(verdict, asked, step)
    except llm.ModelError as error:
        step.update(rules=skipped, status='UNMEASURED', error=str(error))
        return
    rows = result.rows + skipped
    step.update(rules=rows, status=judge.verdict_of(rows), model=result.model, second=second, error=None)
```

```tsx
import { cn } from "@/lib/utils";

const WORD: Record<string, string> = {

  NOT_APPLICABLE: "не к месту",

function Reply({ title, text, status }: { title: string; text: string | null; status?: string }) {
  return (
    <div className="min-w-0 flex-1 rounded-control border border-line p-3">
      <div className="mb-1 text-small text-fg-3">
        {title}
        {status && status !== "200" ? ` · статус ${status}` : ""}

    </div>
  );
}

function Trace({ trace }: { trace: AgentTrace }) {
  return (
    <details className="rounded-control border border-line p-3 text-small">
      <summary className="cursor-pointer text-fg-2">Что происходило внутри агента</summary>
      {trace.rag.map((call, i) => (
        <div key={i} className="mt-3 space-y-2">
          <div>
            <span className="text-fg-3">Запрос в базу знаний: </span>
            {call.query}
          </div>
          <ol className="space-y-1">
            {call.passages.map((p, j) => (
              <li key={j} className="rounded-sm bg-hover p-2">
                <div className="text-fg-3">
                  Статья {String(p.article ?? "—")}, фрагмент {p.passage ?? "—"} · релевантность{" "}
                  {p.reranker ?? p.retrieval ?? "—"}
                </div>
                <p className="whitespace-pre-wrap">{p.text}</p>
              </li>
            ))}
          </ol>
          <div>
            <span className="text-fg-3">Ответ базы знаний: </span>
            {call.answer || call.reason || "—"}
          </div>
        </div>
      ))}
      {trace.rag.length === 0 && <p className="mt-3 text-fg-3">К базе знаний агент не обращался.</p>}
      <ul className="mt-3 space-y-1">
        {trace.chains.map((chain, i) => (
          <li key={i}>
            <span className="text-fg-3">{chain.name}: </span>
            <span className="whitespace-pre-wrap">{chain.error ?? chain.output ?? "—"}</span>
          </li>
        ))}
      </ul>
      {trace.systems.length > 0 && (
        <p className="mt-3">
          <span className="text-fg-3">Системы банка: </span>

function Verdicts({ rules }: { rules: Rule[] }) {
  const shown = rules.filter((r) => r.status !== "NOT_APPLICABLE");
  return (

      {shown.map((r) => (
        <li key={r.ruleId} className="text-small">

            {WORD[r.status] ?? r.status}
          </span>
          <span className="text-fg-3"> · {FAMILY_NAME[familyOf(r.ruleId)]} · </span>
          {r.rule}
          {r.reason && <p className="text-fg-2">{r.reason}</p>}
          {r.agentQuote && <p className="text-fg-3">«{r.agentQuote}»</p>}
        </li>
      ))}
    </ul>
  );
}

      </div>
      {step.trace && <Trace trace={step.trace} />}
```

```tsx
import { useState } from "react";
import { Header } from "../../app/Header";
import { SectionJob } from "../../app/SectionJob";
import { api } from "../../lab/api";
import { pct } from "../../lab/format";
import { useLabState } from "../../lab/LabProvider";
import { FAMILIES, FAMILY_NAME, useReplay } from "../../lab/replay";
import { Button } from "../../ui/Button";
import { EmptyState } from "../../ui/EmptyState";
import { useToast } from "../../ui/toast";
import { Dot } from "../simulations/parts";
import { StepView } from "./StepView";

export function ReplayPage() {
  const { state, refresh } = useLabState();
  const toast = useToast();
  const replay = useReplay(state);
  const targets = state?.targets ?? [];
  const [target, setTarget] = useState(() => targets.find((t) => t.kind === "code")?.id ?? targets[0]?.id ?? "");
  const [count, setCount] = useState(10);
  const [open, setOpen] = useState<string | null>(null);
  const running = state?.job.running && state.job.kind === "replay";
  const result = replay.data && "id" in replay.data ? replay.data : null;

  const start = () =>
    api("/api/replay", { target, count })
      .then(refresh)
      .catch(toast.error);

  return (
    <>
      <Header
        title="Повтор логов"
        sub="Разговоры из выгрузки проходят через агента заново: на каждом шаге видно, что он искал в базе знаний, что нашёл и что ответил."
        below={<SectionJob kinds={["replay"]} />}
      />
      <div className="space-y-6 p-4 lg:p-5">
        <div className="flex flex-wrap items-end gap-3">
          <label className="text-small text-fg-3">
            Агент
            <select className="mt-1 block rounded-control border border-line bg-bg p-2 text-body" value={target} onChange={(e) => setTarget(e.target.value)}>
              {targets.map((t) => (
                <option key={t.id} value={t.id}>
                  {t.name}
                </option>
              ))}
            </select>
          </label>
          <label className="text-small text-fg-3">
            Разговоров
            <input
              type="number"
              min={1}
              max={200}
              className="mt-1 block w-24 rounded-control border border-line bg-bg p-2 text-body"
              value={count}
              onChange={(e) => setCount(Number(e.target.value))}
            />
          </label>
          <Button variant="primary" onClick={start} disabled={!target || running}>
            Повторить
          </Button>
        </div>

        {!result && <EmptyState title="Повторов ещё не было">Выберите локального агента и запустите повтор.</EmptyState>}

        {result && (
          <>
            <div className="grid grid-cols-1 gap-3 sm:grid-cols-3">
              {FAMILIES.map((family) => {
                const score = result.metric[family];
                return (
                  <div key={family} className="rounded-control border border-line p-3">
                    <div className="text-small text-fg-3">{FAMILY_NAME[family]}</div>
                    <div className="text-title tabular-nums">{score.accuracy === null ? "—" : pct(score.accuracy)}</div>
                    <div className="text-small text-fg-3">
                      выполнено {score.pass} из {score.pass + score.fail}
                    </div>
                  </div>
                );
              })}
            </div>
            <ul>
              {result.dialogues.map((d) => (
                <li key={d.dialogueId} className="border-t border-line">
                  <button
                    type="button"
                    className="flex w-full items-center gap-2 py-3 text-left text-body"
                    onClick={() => setOpen(open === d.dialogueId ? null : d.dialogueId)}
                  >
                    <Dot status={d.status} />
                    <span className="min-w-0 flex-1 truncate">{d.steps[0]?.customer}</span>
                    <span className="text-small text-fg-3">шагов: {d.steps.length}</span>
                  </button>
                  {open === d.dialogueId && d.steps.map((step) => <StepView key={step.index} step={step} />)}
                </li>
              ))}
            </ul>
          </>
        )}
      </div>
    </>
  );
}
```
